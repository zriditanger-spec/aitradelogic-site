// ✅ FIX LCP - Helper to build Unsplash URL with correct params
function buildUnsplashUrl(baseUrl, width = 800, height = 450, quality = 75) {
    const clean = baseUrl.split('?')[0];
    return `${clean}?auto=format&fit=crop&w=${width}&h=${height}&q=${quality}&fm=webp`;
}

// ✅ FIX LCP - Build srcset string from a base image URL
function buildSrcset(imageUrl) {
    const base = imageUrl.split('?')[0];
    const s400 = buildUnsplashUrl(base, 400, 225, 70);
    const s800 = buildUnsplashUrl(base, 800, 450, 75);
    const s1200 = buildUnsplashUrl(base, 1200, 675, 80);
    return `${s400} 400w, ${s800} 800w, ${s1200} 1200w`;
}

async function fetchGemini(env, payload, usePro = false) {
    const keysString = env.GEMINI_KEYS || env.GEMINI_API_KEY || "";
    const keys = keysString.split(',').map(k => k.trim()).filter(k => k !== "");
    
    if (keys.length === 0) throw new Error("No Gemini API keys found in environment variables.");

    const modelsToTry = usePro 
        ? ["gemini-3.1-pro-preview", "gemini-3-flash-preview", "gemini-3.1-flash-lite-preview", "gemini-2.5-flash-preview"] 
        : ["gemini-3-flash-preview", "gemini-3.1-flash-lite-preview", "gemini-2.5-flash-preview"];
    let lastError = "No response from API";

    for (const modelName of modelsToTry) {
        for (const key of keys) {
            try {
                const url = `https://generativelanguage.googleapis.com/v1beta/models/${modelName}:generateContent?key=${key}`;
                const res = await fetch(url, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });

                if (res.status === 429) {
                    lastError = "Rate limit exceeded (429). Try again in a few seconds.";
                    console.log(`Rate limit hit for key: ${key.substring(0, 6)}... on model ${modelName}.`);
                    continue; 
                }

                const data = await res.json();
                if (!res.ok) {
                    lastError = data.error?.message || `API Error ${res.status}`;
                    console.log(`Gemini API Error (${res.status}) for model ${modelName}:`, lastError);
                    continue;
                }

                return data; 
            } catch (error) {
                lastError = error.message;
                console.log(`Fetch error for key ${key.substring(0, 6)} on model ${modelName}:`, lastError);
                continue;
            }
        }
    }
    throw new Error(`All Gemini API keys and models failed. Last error: ${lastError}`);
}

async function generateAIImage(env, prompt) {
    try {
        const cleanPrompt = prompt.substring(0, 150).replace(/[^\w\s,.?!]/gi, ' ').trim();
        const enhancedPrompt = `Professional financial news photo, editorial style: ${cleanPrompt}. High resolution, 8k, cinematic lighting, sharp focus. No text, no logos.`;
        
        const pollinationsUrl = `https://image.pollinations.ai/prompt/${encodeURIComponent(enhancedPrompt)}?width=1280&height=720&nologo=true`;
        
        console.log("Fetching image from Pollinations...");
        const imageRes = await fetch(pollinationsUrl);
        
        if (!imageRes.ok) throw new Error("Pollinations failed to generate image");
        
        const imageBlob = await imageRes.blob();
        
        if (env.IMGBB_API_KEY) {
            console.log("Uploading to ImgBB...");
            const formData = new FormData();
            formData.append("image", imageBlob, "image.jpg");
            
            const uploadRes = await fetch(`https://api.imgbb.com/1/upload?key=${env.IMGBB_API_KEY}`, {
                method: 'POST',
                body: formData
            });
            const uploadData = await uploadRes.json();
            
            if (uploadData.success) {
                return { url: uploadData.data.url, source: 'pollinations-ai (ImgBB)' };
            } else {
                console.log("ImgBB Upload Failed:", uploadData.error?.message);
            }
        }
        
        return { url: pollinationsUrl, source: 'pollinations-ai' };
        
    } catch (e) {
        console.log("AI Image Error:", e.message);
        return { url: null, source: "failed_ai", error: e.message };
    }
}

async function processAutoPredictions(env) {
    console.log("Fetching live prices for predictions...");
    const coins = [
        { symbol: 'BTCUSDT', name: 'Bitcoin', coin: 'BTC' },
        { symbol: 'ETHUSDT', name: 'Ethereum', coin: 'ETH' },
        { symbol: 'SOLUSDT', name: 'Solana', coin: 'SOL' },
        { symbol: 'BNBUSDT', name: 'BNB', coin: 'BNB' },
        { symbol: 'XRPUSDT', name: 'XRP', coin: 'XRP' },
        { symbol: 'ADAUSDT', name: 'Cardano', coin: 'ADA' },
        { symbol: 'DOGEUSDT', name: 'Dogecoin', coin: 'DOGE' },
        { symbol: 'AVAXUSDT', name: 'Avalanche', coin: 'AVAX' },
        { symbol: 'LINKUSDT', name: 'Chainlink', coin: 'LINK' },
        { symbol: 'MATICUSDT', name: 'Polygon', coin: 'MATIC' }
    ];
    let livePrices = {};
    for (const c of coins) {
        try {
            const res = await fetch(`https://api.binance.com/api/v3/ticker/price?symbol=${c.symbol}`);
            const data = await res.json();
            livePrices[c.coin] = parseFloat(data.price).toFixed(4);
        } catch(e) { console.log("Failed to fetch price for", c.symbol); }
    }

    const prompt = `Act as a Quantitative Trading Bot.
Analyze these 10 cryptos based on LIVE PRICES: ${JSON.stringify(livePrices)}.
Provide a 24h technical forecast (Entry, Target, StopLoss).
Return ONLY a valid JSON array.
Format: [{"coin":"BTC","name":"Bitcoin","price":"...","trend":"Bullish","entry":"...","target":"...","sl":"...","confidence":"85"}]`;

    try {
        const geminiPayload = { 
            contents: [{ parts: [{ text: prompt }] }], 
            generationConfig: { temperature: 0.2, responseMimeType: "application/json" } 
        };
        const geminiData = await fetchGemini(env, geminiPayload, false);
        let rawText = geminiData.candidates[0].content.parts[0].text.replace(/```json/g, "").replace(/```/g, "").trim();
        const predictionsJSON = JSON.parse(rawText);
        if (predictionsJSON && predictionsJSON.length > 0) {
            await env.DB.prepare("INSERT INTO settings (key, value) VALUES ('daily_predictions', ?) ON CONFLICT(key) DO UPDATE SET value=excluded.value").bind(JSON.stringify(predictionsJSON)).run();
            console.log("Auto Predictions updated successfully.");
        }
    } catch(e) { console.log("Auto Predictions Error:", e.message); }
}

// ✅ FIX LCP #1 — getDynamicImage: URL propre, paramètres corrects, WebP forcé
async function getDynamicImage(searchKeyword, env) {
    let category = 'crypto';

    if (env.UNSPLASH_API_KEY) {
        try {
            const query = encodeURIComponent(searchKeyword.substring(0, 30));
            const res = await fetch(`https://api.unsplash.com/photos/random?query=${query}&orientation=landscape&client_id=${env.UNSPLASH_API_KEY}`);
            
            if (res.status === 403 || res.status === 429) {
                console.log("Unsplash Rate Limit! Switching to fallback silently.");
            } else if (res.ok) {
                const data = await res.json();
                if (data && data.urls && data.urls.raw) {
                    // ✅ URL propre: w=800, h=450, WebP, quality=75 — pas de double paramètre
                    return buildUnsplashUrl(data.urls.raw, 800, 450, 75);
                }
            }
        } catch(e) {
            console.error("Unsplash error:", e.message);
        }
    }
    
    const images = FALLBACK_IMAGES[category];
    const fallbackImg = images[Math.floor(Math.random() * images.length)];
    // ✅ Fallback aussi en WebP optimisé
    return buildUnsplashUrl(fallbackImg, 800, 450, 75);
}

function escapeRegExp(string) { return string.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'); }

function createSafeSlug(text) {
    if (!text) return "post";
    return text.toLowerCase().replace(/[^a-z0-9\u0600-\u06FF\u00C0-\u017F-]/g, '-').replace(/-+/g, '-').replace(/^-|-$/g, '');
}

function applySmartLinks(htmlContent, glossaryList, blogList, currentSlug = "") {
    if (!htmlContent) return htmlContent;
    let parts = htmlContent.split(/(<[^>]+>)/g);
    let usedTerms = new Set();
    let linkableItems = [];

    if (glossaryList && glossaryList.length > 0) {
        glossaryList.forEach(item => {
            if (item.term.length > 2) linkableItems.push({ term: item.term, url: '/glossary.html', css: 'text-emerald-400 font-bold hover:underline transition-colors' });
        });
    }

    if (blogList && blogList.length > 0) {
        blogList.filter(post => post.id !== currentSlug).slice(0, 10).forEach(post => {
            let firstWord = post.title.split(' ').find(w => w.length > 4); 
            if (firstWord) {
                let cleanWord = firstWord.replace(/[^a-zA-Z0-9]/g, '');
                linkableItems.push({ term: cleanWord, url: `/blog/${post.id}`, css: 'text-purple-400 font-bold hover:underline transition-colors' });
            }
        });
    }

    linkableItems.sort((a, b) => b.term.length - a.term.length);

    linkableItems.forEach(item => {
        let lowerTerm = item.term.toLowerCase();
        if (usedTerms.has(lowerTerm)) return;
        const regex = new RegExp('(^|[^a-zA-Z0-9_])(' + escapeRegExp(item.term) + ')([^a-zA-Z0-9_]|$)', 'i');
        let replaced = false;

        for (let i = 0; i < parts.length; i++) {
            if (i % 2 === 0 && !replaced && !parts[i].includes('</a>')) {
                if (regex.test(parts[i])) {
                    parts[i] = parts[i].replace(regex, `$1<a href="${item.url}" class="${item.css}" target="_blank" title="Read more">$2</a>$3`);
                    usedTerms.add(lowerTerm);
                    replaced = true; 
                }
            }
        }
    });
    return parts.join('');
}

// Helper function for Google Indexing JWT
async function getGoogleAccessToken(clientEmail, privateKey) {
    const header = { alg: 'RS256', typ: 'JWT' };
    const now = Math.floor(Date.now() / 1000);
    const claim = {
        iss: clientEmail,
        scope: 'https://www.googleapis.com/auth/indexing',
        aud: 'https://oauth2.googleapis.com/token',
        exp: now + 3600,
        iat: now
    };

    const base64url = (str) => btoa(str).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
    const encodedHeader = base64url(JSON.stringify(header));
    const encodedClaim = base64url(JSON.stringify(claim));
    const unsignedToken = `${encodedHeader}.${encodedClaim}`;

    const pemContents = privateKey.replace(/-----BEGIN PRIVATE KEY-----/g, '').replace(/-----END PRIVATE KEY-----/g, '').replace(/\s/g, '');
    const binaryDerString = atob(pemContents);
    const binaryDer = new Uint8Array(binaryDerString.length);
    for (let i = 0; i < binaryDerString.length; i++) {
        binaryDer[i] = binaryDerString.charCodeAt(i);
    }

    const key = await crypto.subtle.importKey(
        'pkcs8',
        binaryDer.buffer,
        { name: 'RSASSA-PKCS1-v1_5', hash: 'SHA-256' },
        false,
        ['sign']
    );

    const signature = await crypto.subtle.sign(
        'RSASSA-PKCS1-v1_5',
        key,
        new TextEncoder().encode(unsignedToken)
    );

    const encodedSignature = base64url(String.fromCharCode(...new Uint8Array(signature)));
    const jwt = `${unsignedToken}.${encodedSignature}`;

    const response = await fetch('https://oauth2.googleapis.com/token', {
        method: 'POST',
        headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
        body: `grant_type=urn%3Aietf%3Aparams%3Aoauth%3Agrant-type%3Ajwt-bearer&assertion=${jwt}`
    });

    const data = await response.json();
    return data.access_token;
}

async function broadcastToSocials(env, title, excerpt, url, imageUrl, twitterPost = null, facebookPost = null, fullContent = null) {
    try {
        if (env.TELEGRAM_BOT_TOKEN && env.TELEGRAM_CHAT_ID) {
            const tgMsg = `🚨 *BREAKING NEWS* 🚨\n\n*${title}*\n\n${excerpt}\n\n[Lire la suite](${url})`;
            try {
                if (imageUrl) {
                    await fetch(`https://api.telegram.org/bot${env.TELEGRAM_BOT_TOKEN}/sendPhoto`, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({ chat_id: env.TELEGRAM_CHAT_ID, photo: imageUrl, caption: tgMsg, parse_mode: 'Markdown' })
                    });
                } else {
                    await fetch(`https://api.telegram.org/bot${env.TELEGRAM_BOT_TOKEN}/sendMessage`, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({ chat_id: env.TELEGRAM_CHAT_ID, text: tgMsg, parse_mode: 'Markdown' })
                    });
                }
            } catch (tgErr) { console.error("Telegram Fetch Error:", tgErr.message); }
        }

        if (env.DISCORD_WEBHOOK_URL) {
            await fetch(env.DISCORD_WEBHOOK_URL, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    embeds: [{ title: title, description: excerpt, url: url, color: 3447003, image: { url: imageUrl } }]
                })
            });
        }

        if (env.MAKE_WEBHOOK_URL) {
            let finalTwitterPost = twitterPost || excerpt || "";
            if (finalTwitterPost.length > 200) finalTwitterPost = finalTwitterPost.substring(0, 200) + "...";

            await fetch(env.MAKE_WEBHOOK_URL, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    title, excerpt, 
                    twitterExcerpt: finalTwitterPost, 
                    facebookExcerpt: facebookPost || excerpt,
                    fullContent: fullContent || excerpt,
                    teaserContent: fullContent ? (fullContent.substring(0, 800) + "...") : excerpt,
                    url, imageUrl,
                    thumbnail: imageUrl,
                    timestamp: new Date().toISOString()
                })
            });
        }

        if (env.GOOGLE_SERVICE_ACCOUNT_JSON) {
            try {
                const credentials = JSON.parse(env.GOOGLE_SERVICE_ACCOUNT_JSON);
                const token = await getGoogleAccessToken(credentials.client_email, credentials.private_key);
                const indexRes = await fetch('https://indexing.googleapis.com/v3/urlNotifications:publish', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${token}` },
                    body: JSON.stringify({ url: url, type: 'URL_UPDATED' })
                });
                console.log(`Google Indexing API status for ${url}:`, indexRes.status);
            } catch (err) {
                console.error("Google Indexing failed:", err.message);
            }
        } else {
            await fetch(`https://www.google.com/ping?sitemap=${url}`);
            console.log(`Pinged Google for ${url} (No Service Account JSON)`);
        }
        
    } catch (e) { console.log("Broadcast failed:", e.message); }
}

        console.log(`Daily limit reached (${dailyCount}/5). Skipping news generation.`);

    }

    const feeds = [

        const { results } = await env.DB.prepare("SELECT data FROM articles WHERE type = 'news' ORDER BY created_at DESC LIMIT 20").all();

            let d = JSON.parse(r.data);
            return (d.EN?.title || d.title || "").toLowerCase();
        });
    } catch(e) { console.log("Failed to fetch recent titles for deduplication"); }

    function isSimilar(title1, title2) {
        if (!title1 || !title2) return false;
        const words1 = title1.split(' ').filter(w => w.length > 3);
        let matches = 0;
        words1.forEach(w => { if (title2.includes(w)) matches++; });

                for (let itemXml of items) {
                    const extractTag = (xml, tag) => {
                        const match = xml.match(new RegExp(`<${tag}[^>]*>(?:<!\\[CDATA\\[)?([\\s\\S]*?)(?:\\]\\]>)?<\\/${tag}>`, 'i'));

                    const pubDateStr = extractTag(itemXml, 'pubDate') || extractTag(itemXml, 'updated') || extractTag(itemXml, 'published');
                    let pubDate = new Date();

                    let tempTitle = extractTag(itemXml, 'title');
                    
                    let isDuplicate = false;
                    for (let oldTitle of recentTitles) {
                        if (isSimilar(tempTitle.toLowerCase(), oldTitle)) { isDuplicate = true; break; }
                    }
                    if (isDuplicate) { console.log("Skipping duplicate news:", tempTitle); continue; }

                    let tempDesc = extractTag(itemXml, 'description').replace(/<[^>]+>/g, '').substring(0, 800);
                    if (!tempDesc) tempDesc = extractTag(itemXml, 'content');
                    
                    if (tempTitle && tempDesc) {

                }

    let imageSource = "unsplash";

    const dateStr = new Date().toLocaleDateString('en-US', { year: 'numeric', month: 'short', day: 'numeric' });

    const groupLinks = {};
    let englishSlug = "";
    for (const lang of ['EN', 'FR', 'ES', 'AR']) {
        if(!allLangs[lang]) continue;
        const slug = createSafeSlug(allLangs[lang].title) + "-" + uniqueId + "_" + lang;
        groupLinks[lang] = slug;
        if (lang === 'EN') englishSlug = slug;
    }

    for (const lang of ['EN', 'FR', 'ES', 'AR']) {
        if(!allLangs[lang]) continue;
        const fullSlug = groupLinks[lang];
        const article = {
            id: fullSlug,
            title: allLangs[lang].title,
            excerpt: allLangs[lang].excerpt,
            content: allLangs[lang].content,
            image: sharedImage,
            image_source: imageSource, 
            date: dateStr,
            category: "Market Insights",
            tags: allLangs[lang].tags || [],
            faq: allLangs[lang].faq || [],
            groupLinks: groupLinks,
            symbol: "none"
        };
        await env.DB.prepare("INSERT INTO articles (slug, type, lang, data) VALUES (?, 'blog', ?, ?) ON CONFLICT(slug) DO UPDATE SET data=excluded.data")
                    .bind(fullSlug, lang, JSON.stringify(article)).run();
    }
    
    if (allLangs['EN'] && englishSlug) {
        const articleUrl = `https://aitradelogic.com/article/${englishSlug}`; 
        await broadcastToSocials(env, allLangs['EN'].title, allLangs['EN'].excerpt, articleUrl, sharedImage, allLangs['EN'].twitter_post, allLangs['EN'].facebook_post, allLangs['EN'].content);
    }
    
    return true;

            for (let i = 0; i < dateStr.length; i++) seed += dateStr.charCodeAt(i);
            for (let i = 0; i < seedStr.length; i++) seed += seedStr.charCodeAt(i);
            
            let blockSize = 24 / count;
            for (let i = 0; i < count; i++) {
                seed = (seed * 9301 + 49297) % 233280;
                let rnd = seed / 233280;
                let hourInBlock = Math.floor(rnd * blockSize);
                hours.push(Math.floor(i * blockSize) + hourInBlock);
            }
            return hours;
        }

        const predSchedule = getDailySchedule(today, 2, "predictions");
        ctx.waitUntil((async () => {
            try {
                if (currentHour === predSchedule[0] || currentHour === predSchedule[1]) {
                    console.log(`Time to generate Predictions!`);
                    await processAutoPredictions(env);
                }
            } catch(e) { console.log("Prediction Error:", e.message); }
        })());

        const newsSchedule = getDailySchedule(today, 4, "news");
        const articleSchedule = getDailySchedule(today, 4, "article");

        console.log(`Today's News Schedule (UTC): ${newsSchedule.join(', ')}`);
        console.log(`Today's Article Schedule (UTC): ${articleSchedule.join(', ')}`);

        ctx.waitUntil((async () => {
            try {
                const { results } = await env.DB.prepare("SELECT COUNT(*) as count FROM articles WHERE type = 'news' AND date(created_at) = ?").bind(today).all();
                let dailyNewsCount = (results && results.length > 0) ? Math.floor(results[0].count / 4) : 0;
                if (dailyNewsCount < 5 && currentHour >= newsSchedule[dailyNewsCount]) {
                    console.log(`Time to generate News! (Count: ${dailyNewsCount}, Target Hour: ${newsSchedule[dailyNewsCount]})`);
                    await processAutoNews(env);
                } else {
                    console.log(`Skipping News. Current Hour: ${currentHour}, Next Target: ${newsSchedule[dailyNewsCount]}`);
                }
            } catch(e) { console.log("News Error:", e.message); }
        })());

        ctx.waitUntil((async () => {
            try {
                const { results } = await env.DB.prepare("SELECT COUNT(*) as count FROM articles WHERE type = 'blog' AND date(created_at) = ?").bind(today).all();
                let dailyArticleCount = (results && results.length > 0) ? Math.floor(results[0].count / 4) : 0;
                if (dailyArticleCount < 4 && currentHour >= articleSchedule[dailyArticleCount]) {
                    console.log(`Time to generate Article! (Count: ${dailyArticleCount}, Target Hour: ${articleSchedule[dailyArticleCount]})`);
                    await processAutoArticle(env);
                } else {
                    console.log(`Skipping Article. Current Hour: ${currentHour}, Next Target: ${articleSchedule[dailyArticleCount]}`);
                }
            } catch(e) { console.log("Article Error:", e.message); }
        })());
    }
